cmake_minimum_required(VERSION 3.20)
project(BilateralTeleoperation LANGUAGES CXX)

include(CTest)
add_library(teleop_support INTERFACE)
target_include_directories(teleop_support INTERFACE core/include)
target_compile_features(teleop_support INTERFACE cxx_std_17)
add_executable(support_tests tests/support_tests.cpp)
target_link_libraries(support_tests PRIVATE teleop_support)
if(MSVC)
    target_compile_options(support_tests PRIVATE /W4 /WX)
else()
    target_compile_options(support_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME support_regressions COMMAND support_tests)
